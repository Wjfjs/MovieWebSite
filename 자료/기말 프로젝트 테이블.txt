create table users (
    ID VARCHAR2(20) PRIMARY KEY,
    PASSWD VARCHAR2(20) NOT NULL,
    EMAIL VARCHAR2(30) NOT NULL,
    CONSTRAINT idLen_users CHECK (LENGTH(ID) BETWEEN 4 and 20),
    CONSTRAINT passwdLen_users CHECK (LENGTH(PASSWD) BETWEEN 6 and 20),
    CONSTRAINT checkEmail_users CHECK (REGEXP_LIKE(EMAIL, '^[A-Za-z0-9]+@[A-Za-z]+\.com$'))
);

create table member (
    MEMBER_ID VARCHAR2(3) PRIMARY KEY,
    CHECK_LOGIN CHAR(1) DEFAULT 'N',
    LOGIN_DATE DATE DEFAULT SYSDATE,
    USER_ID VARCHAR2(20) NOT NULL,
    LIKE_MOVIE_NO VARCHAR2(100) DEFAULT '0',
    MANAGE_MEMBER_ID VARCHAR2(15) DEFAULT '000',
    CONSTRAINT checkLogin_member CHECK (CHECK_LOGIN IN ('Y', 'N')),
    CONSTRAINT idLen_users_member CHECK (LENGTH(USER_ID) BETWEEN 4 and 20),
    CONSTRAINT USER_ID_FK_member FOREIGN KEY (USER_ID) REFERENCES users (ID),
    CONSTRAINT MANAGE_MEMBER_ID_FK FOREIGN KEY (MANAGE_MEMBER_ID) REFERENCES member (MEMBER_ID)
);

CREATE SEQUENCE member_id_seq START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER member_id_trigger
BEFORE INSERT ON member
FOR EACH ROW
DECLARE
  next_id NUMBER;
BEGIN
  SELECT NVL(MAX(TO_NUMBER(MEMBER_ID)), 0) + 1 INTO next_id FROM member;
  :NEW.MEMBER_ID := LPAD(next_id, 3, '0');
END;
/

create table movie (
    NO NUMBER PRIMARY KEY,
    KR_NAME VARCHAR2(100) DEFAULT 'N/A',
    EN_NAME VARCHAR2(100) DEFAULT 'N/A',
    DRCTR VARCHAR2(100) DEFAULT 'N/A',
    MAKR VARCHAR2(100) DEFAULT 'N/A',
    INCME VARCHAR2(100) DEFAULT 'N/A',
    DISTB VARCHAR2(100) DEFAULT 'N/A',
    MK_DATE VARCHAR2(10) DEFAULT 'N/A',
    COUNTRY CHAR(3) DEFAULT 'N/A',
    SCRN_COUNT NUMBER DEFAULT 0,
    VIEW_COUNT NUMBER DEFAULT 0,
    GENRE VARCHAR2(200) DEFAULT 'N/A',
    GRAND CHAR(6) DEFAULT 'na.png',
    POSTER VARCHAR2(300)DEFAULT 'NA.jpg',
    TRAILER VARCHAR2(300) DEFAULT 'NA.jpg',
    STORY VARCHAR2(2000) DEFAULT 'N/A',
    RUN_TIME NUMBER(3) DEFAULT 0,
    LIKE_NUM NUMBER DEFAULT 0,
    OPEN_NUM NUMBER DEFAULT 0,
    MANAGE_MEMBER_ID VARCHAR2(3) DEFAULT '000',
    CONSTRAINT MANAGE_MEMBER_ID_FK_movie FOREIGN KEY (MANAGE_MEMBER_ID) REFERENCES member (MEMBER_ID)
);

CREATE SEQUENCE movie_no_seq START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER movie_trigger
BEFORE INSERT ON movie
FOR EACH ROW
DECLARE
  next_id NUMBER;
BEGIN
  SELECT NVL(MAX(NO), 0) + 1 INTO next_id FROM movie;
  :NEW.NO := next_id;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    :NEW.NO := 1;
END;
/

BEGIN

/
